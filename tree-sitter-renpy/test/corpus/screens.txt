==================
vbox/hbox nesting with for/if, use with arguments, key, timer and imagebutton (nms4.3)
==================
screen inventory(items):
    vbox:
        spacing 10
        for item in items:
            hbox:
                text item.name
                if item.count > 1:
                    text "x[item.count]" size 14
                else:
                    null width 10
    use item_tooltip(items[0], show_price=True)
    use navigation
    key "K_ESCAPE" action Hide("inventory")
    timer 5.0 repeat True action Function(tick)
    imagebutton:
        idle "close_idle.png"
        hover "close_hover.png"
        action Return()
    imagebutton idle "a.png" hover "b.png" action Jump("start") xalign 1.0
