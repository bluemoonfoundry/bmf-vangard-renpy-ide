==================
Trailing comments on simple statements in a label (nms4.27)
==================
label start:
    jump b # c
    call x # c
    show e # c
    show e at left with dissolve # c
    hide e # c
    scene bg # c
    with fade # c
    stop music # c
    voice "v.ogg" # c
    play music "a.ogg" # c
    nvl clear # c
    pass # c
    $ x = 1 # c
    e "hi" # c
    "Narration." # c
    return # c

---

(source_file
  (label_statement
    name: (identifier)
    (jump_statement target: (identifier) (comment))
    (call_statement target: (identifier) (comment))
    (show_statement image: (identifier) (comment))
    (show_statement
      image: (identifier)
      at_expr: (expression)
      transition: (expression)
      (comment))
    (hide_statement image: (identifier) (comment))
    (scene_statement image: (identifier) (comment))
    (with_statement transition: (expression) (comment))
    (stop_statement channel: (identifier) (comment))
    (voice_statement file: (string) (comment))
    (play_statement channel: (identifier) file: (string) (comment))
    (nvl_statement (comment))
    (pass_statement (comment))
    (dollar_statement expression: (expression) (comment))
    (say_statement character: (identifier) text: (string) (comment))
    (say_statement text: (string) (comment))
    (return_statement (comment))))
